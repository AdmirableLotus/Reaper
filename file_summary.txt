    function animateValue(el, value) {

    async function fetchSupply() {
      const url = `https://api.etherscan.io/v2/api?chain=${CHAIN}&module=stats&action=tokensupply&contractaddress=${CONTRACT}&apikey=${API_KEY}`;
      const res = await fetch(url);
      const data = await res.json();

      if (!data.result) return "-";

      const supply = (parseInt(data.result) / 1e18).toLocaleString();
      animateValue(document.getElementById("totalSupply"), supply + " RPR");
    }

    async function fetchHolders() {
      const url = `https://api.etherscan.io/v2/api?chain=${CHAIN}&module=token&action=tokenholderlist&contractaddress=${CONTRACT}&page=1&offset=200&apikey=${API_KEY}`;
      const res = await fetch(url);
      const data = await res.json();

      const count = data.result?.length || 0;
      animateValue(
        document.getElementById("holders"),
        count + " Survivor" + (count === 1 ? "" : "s")
      );

      if (!window.ethereum) {
        animateValue(document.getElementById("walletBalance"), "Install MetaMask");
        return;
      }

      const accounts = await window.ethereum.request({
        method: "eth_requestAccounts",
      });

      const wallet = accounts[0];

      const url = `https://api.etherscan.io/v2/api?chain=${CHAIN}&module=account&action=tokenbalance&contractaddress=${CONTRACT}&address=${wallet}&tag=latest&apikey=${API_KEY}`;

      const res = await fetch(url);
      const data = await res.json();

      if (!data.result) return;

      const balance = (parseInt(data.result) / 1e18).toLocaleString();
      animateValue(document.getElementById("walletBalance"), balance + " RPR");
    }
